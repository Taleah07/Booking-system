<script>
    function toggleEditMode() {
        //Find all edit links and header controls
        const editLinks = document.querySelectorAll('.edit-links');
        const headerControls = document.querySelectorAll('.header-controls');
        const weekEditDivs = document.querySelectorAll('td.week-col > div');
        const button = document.getElementById('toggleEditBtn');

        editLinks.forEach(element => {
            const currentDisplay = window.getComputedStyle(element).display;
            if (currentDisplay === 'none' || currentDisplay === '') {
                element.style.setProperty('display', 'flex', 'important');
            } else {
                element.style.setProperty('display', 'none', 'important');
            }
        });

        headerControls.forEach(element => {
            const currentDisplay = window.getComputedStyle(element).display;
            if (currentDisplay === 'none' || currentDisplay === '') {
                element.style.setProperty('display', 'flex', 'important');
            } else {
                element.style.setProperty('display', 'none', 'important');
            }
        });

        weekEditDivs.forEach(element => {
            const currentDisplay = window.getComputedStyle(element).display;
            if (currentDisplay === 'none' || currentDisplay === '') {
                element.style.setProperty('display', 'flex', 'important');
            } else {
                element.style.setProperty('display', 'none', 'important');
            }
        });

        //update the master button text dynamically
        if (button.innerText.trim() === 'Enable Editing') {
            button.innerText = 'Disable Editing';
            button.classList.replace('btn-warning','btn-secondary');
        } else {
            button.innerText = 'Enable Editing';
            button.classList.replace('btn-warning','btn-secondary');
        }
    }
</script>
